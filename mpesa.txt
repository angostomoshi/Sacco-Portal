CREATE TABLE public.pb_mpesa_transactions (
	id serial4 NOT NULL,
	merchant_request_id varchar(100) NULL,
	checkout_request_id varchar(100) NOT NULL,
	member_no varchar(50) NOT NULL,
	purpose varchar(20) NOT NULL,
	account_reference varchar(50) NOT NULL,
	phone_no varchar(15) NOT NULL,
	amount numeric(14, 2) NOT NULL,
	status varchar(20) DEFAULT 'pending'::character varying NOT NULL,
	result_code int4 NULL,
	result_desc text NULL,
	mpesa_receipt_no varchar(30) NULL,
	transaction_date timestamp NULL,
	callback_raw jsonb NULL,
	ledger_posted bool DEFAULT false NOT NULL,
	ledger_posted_at timestamp NULL,
	receipt_emailed bool DEFAULT false NOT NULL,
	sms_sent bool DEFAULT false NOT NULL,
	created_at timestamp DEFAULT now() NOT NULL,
	updated_at timestamp DEFAULT now() NOT NULL,
	receipted bool DEFAULT false NULL,
	tno varchar(50) NULL,
	receipt_no varchar(50) DEFAULT 'N/A'::character varying NULL,
	CONSTRAINT pb_mpesa_transactions_checkout_request_id_key UNIQUE (checkout_request_id),
	CONSTRAINT pb_mpesa_transactions_pkey PRIMARY KEY (id)
);
CREATE INDEX idx_mpesa_txn_member ON public.pb_mpesa_transactions USING btree (member_no);
CREATE INDEX idx_mpesa_txn_status ON public.pb_mpesa_transactions USING btree (status);
